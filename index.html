let calYear, calMonth; // calMonth: 0-11
let dateFieldTouched = false;
let searchTerm = '';
function initCalDate(){
const now = new Date();
calYear = now.getFullYear();
calMonth = now.getMonth();
}
function syncDateFieldToCalendar(){
const dateInput = document.getElementById('f_date');
if(!dateInput || dateFieldTouched) return;
const mm = String(calMonth+1).padStart(2,'0');
dateInput.value = `${calYear}-${mm}-01`;
}
function renderTodaySchedule(){
const el = document.getElementById('todaySchedule');
const dateEl = document.getElementById('todayDateLabel');
if(!el || !dateEl) return; // 오늘의 일정 패널을 제거한 레이아웃에서는 여기서 조용히 종료
const today = todayStr();
const d = parseYMD(today);
dateEl.textContent = `${d.getMonth()+1}월 ${d.getDate()}일 (${dowNames[d.getDay()]})`;
const items = bookings
.filter(b => b.date === today && b.status !== 'rejected')
.sort((a,b) => a.start.localeCompare(b.start));
if(!items.length){
el.innerHTML = '<div class="today-empty">오늘 등록된 일정이 없습니다.</div>';
return;
}
el.innerHTML = items.map(b => `
<div class="today-item">
<div class="today-time">
<span class="today-dot ${b.status}"></span>${escapeHtml(b.start)} – ${escapeHtml(b.end)}
</div>
<div class="today-loc">${escapeHtml(b.location || '미지정')}</div>
<div class="today-purpose">${escapeHtml(b.purpose || '')} <span style="color:var(--muted);">(${escapeHtml(b.requester || '')})</span></div>
</div>
`).join('');
}
function renderStats(){
const total = bookings.length;
const pending = bookings.filter(b=>b.status==='pending').length;
const approved = bookings.filter(b=>b.status==='approved').length;
const rejected = bookings.filter(b=>b.status==='rejected').length;
document.getElementById('statTotal').textContent = total;
document.getElementById('statPending').textContent = pending;
document.getElementById('statApproved').textContent = approved;
document.getElementById('statRejected').textContent = rejected;
document.getElementById('statPendingChip').classList.toggle('has-pending', pending > 0);
}
function render(){
const listEl = document.getElementById('scheduleList');
const countEl = document.getElementById('countLabel');
document.getElementById('calTitle').textContent = `${calYear}년 ${calMonth+1}월`;
syncDateFieldToCalendar();
renderStats();
if(bookings.length === 0){
countEl.textContent = '';
listEl.innerHTML = '<div class="empty">등록된 일정이 없어요. 왼쪽에서 첫 일정을 등록해 보세요.</div>';
renderTodaySchedule();
return;
}
const monthBookings = bookings.filter(b=>{
const d = parseYMD(b.date);
return d.getFullYear() === calYear && d.getMonth() === calMonth;
});
countEl.textContent = monthBookings.length ? `이번 달 ${monthBookings.length}건` : '이번 달 일정 없음';
const byDate = {};
monthBookings.forEach(b=>{ (byDate[b.date] = byDate[b.date] || []).push(b); });
Object.values(byDate).forEach(arr => arr.sort((a,b)=> a.start < b.start ? -1 : (a.start > b.start ? 1 : 0)));
const firstOfMonth = new Date(calYear, calMonth, 1);
const startWeekday = firstOfMonth.getDay();
const daysInMonth = new Date(calYear, calMonth+1, 0).getDate();
const totalCells = Math.ceil((startWeekday + daysInMonth) / 7) * 7;
const todayS = todayStr();
let html = `<div class="cal-grid">`;
dowNames.forEach((n,i)=>{
html += `<div class="cal-dow ${i===0?'sun':i===6?'sat':''}">${n}</div>`;
});
for(let i=0;i<totalCells;i++){
const dayNum = i - startWeekday + 1;
let cellDate, outside = false;
if(dayNum < 1){
const prevMonthDays = new Date(calYear, calMonth, 0).getDate();
cellDate = new Date(calYear, calMonth-1, prevMonthDays + dayNum);
outside = true;
}else if(dayNum > daysInMonth){
cellDate = new Date(calYear, calMonth+1, dayNum - daysInMonth);
outside = true;
}else{
cellDate = new Date(calYear, calMonth, dayNum);
}
const dateStr = cellDate.getFullYear()+'-'+String(cellDate.getMonth()+1).padStart(2,'0')+'-'+String(cellDate.getDate()).padStart(2,'0');
const isToday = dateStr === todayS;
const dayEvents = byDate[dateStr] || [];
let eventsHtml = '';
const locGroups = {};
dayEvents.forEach(b=>{
const loc = b.location || '미지정';
(locGroups[loc] = locGroups[loc] || []).push(b);
});
const locOrder = Object.keys(locGroups).sort((a,b)=>a.localeCompare(b,'ko'));
locOrder.forEach(loc=>{
eventsHtml += `<div class="cal-loc-label">${escapeHtml(loc)}</div>`;
locGroups[loc].forEach(b=>{
const conflicts = dayEvents.filter(o => o.id !== b.id && o.status!=='rejected' && b.status!=='rejected' && o.location === b.location && timesOverlap(b.start,b.end,o.start,o.end));
const searchMatch = searchTerm && [b.purpose,b.requester,b.location,b.note].some(f => (f||'').toLowerCase().includes(searchTerm));
eventsHtml += `
<div class="cal-event ${b.status} ${searchMatch ? 'search-match' : ''}">
<span class="ev-time">${b.start}–${b.end}</span>${conflicts.length ? ' <span style="color:var(--rust);">겹침</span>' : ''}<br>
<span class="ev-purpose">${escapeHtml(b.purpose)}</span> <span class="ev-name">(${escapeHtml(b.requester)})</span>
</div>`;
if(isAdmin){
eventsHtml += `<div class="cal-event-actions">
${b.status !== 'approved' ? `<button class="approve-btn" data-id="${b.id}" data-action="approved">승인</button>` : ''}
${b.status !== 'rejected' ? `<button class="reject-btn" data-id="${b.id}" data-action="rejected">반려</button>` : ''}
<button class="del-btn" data-id="${b.id}" data-action="delete">삭제</button>
</div>`;
}
});
